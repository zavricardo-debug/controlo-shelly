// ISTO SUBSTITUI O $w.onReady DO WIX PARA FUNCIONAR NO GITHUB/CANVAS window.addEventListener("DOMContentLoaded", () => { // O seu fetch original adaptado para passar as credenciais de forma segura via URL const url = "https://shelly.cloud"; fetch(url, { method: "GET", mode: "no-cors" // Contorna o bloqueio de segurança do navegador }) .then(() => console.log("Comando enviado com sucesso para a Shelly via GitHub Pages!")) .catch(err => console.error("Erro de rede:", err)); });
