  <canvas id="canvasShelly" width="300" height="120"></canvas>

  <p id="status">Clique no botão acima para acionar a Shelly</p>


  <script>

    const canvas = document.getElementById("canvasShelly");

    const ctx = canvas.getContext("2d");

    const statusTexto = document.getElementById("status");


    // Desenho do botão no Canvas

    const botao = { x: 30, y: 35, largura: 240, altura: 50, cor: "#007bff", texto: "⚡ LIGAR SHELLY" };


    function desenhar() {

      ctx.fillStyle = botao.cor;

      // Desenha retângulo com cantos ligeiramente arredondados nativos

      ctx.beginPath();

      ctx.roundRect ? ctx.roundRect(botao.x, botao.y, botao.largura, botao.altura, 8) : ctx.rect(botao.x, botao.y, botao.largura, botao.altura);

      ctx.fill();

      

      ctx.fillStyle = "#ffffff";

      ctx.font = "bold 16px Arial";

      ctx.textAlign = "center";

      ctx.textBaseline = "middle";

      ctx.fillText(botao.texto, botao.x + (botao.largura / 2), botao.y + (botao.altura / 2));

    }


    function enviarComando() {

      statusTexto.textContent = "A enviar sinal para a Cloud...";

      statusTexto.style.color = "#007bff";

      

      fetch("https://shelly.cloud", { 

        body: "channel=0&turn=on&timer=1&id=441793a5621c&auth_key=ZjJmMGF1aWQE3BBDBB3C3994FEF570BD05536F0B61870FB50B101EC413C1AD36FD450AB5E5381162F50DF2E1458", 

        headers: { "Content-Type": "application/x-www-form-urlencoded" }, 

        method: "POST"

      })

      .then(res => res.json())

      .then(data => {

        console.log(data);

        statusTexto.textContent = "🟢 Sucesso! Comando registado na Shelly.";

        statusTexto.style.color = "#28a745";

      })

      .catch(err => {

        console.error(err);

        statusTexto.textContent = "🔴 O navegador bloqueou o pedido (CORS).";

        statusTexto.style.color = "#dc3545";

      });

    }


    canvas.addEventListener("click", (e) => {

      const rect = canvas.getBoundingClientRect();

      const x = e.clientX - rect.left;

      const y = e.clientY - rect.top;

      if (x >= botao.x && x <= botao.x + botao.largura && y >= botao.y && y <= botao.y + botao.altura) {

        enviarComando();

      }

    });


    desenhar();

  </script>
